  function syncCronExactPreview(cronStr) {
    const s = (cronStr || '').trim();
    if (routineCronExactValue) routineCronExactValue.textContent = s || '(empty)';
    if (routineHumanPreview) {
      const label = getHumanCronPreview(s);
      const span = routineHumanPreview.querySelector('span');
      if (span) span.textContent = `Schedule: ${label}`;
      else routineHumanPreview.textContent = `Schedule: ${label}`;
    }
  }

  async function fetchAllAgents() {
    try {
      const res = await fetch('/api/agents');
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const payload = await res.json();
      const agents = agentsFromApiPayload(payload);
      state.agents = agents;
      return agents;
    } catch (err) {
      console.error('[AutoReiv Routines] Failed to fetch /api/agents:', err);
      return agentsFromApiPayload(state.agents);
    }
  }

  /** CARD-310: filter agent dropdown from full /api/agents — not routine agent_ids. */
  async function populateRoutinesFilterAgents() {
    if (!routinesFilterAgent) return;
    const prev = routinesFilterAgent.value;
    const agents = await fetchAllAgents();
    routinesFilterAgent.innerHTML = buildFilterAgentOptionsHtml(agents);
    const ids = agents.map((a) => a.id);
    if (prev && ids.includes(prev)) routinesFilterAgent.value = prev;
  }

  function showRoutineBanner(msg, isError = false) {
    if (!routineStatusBanner) return;
    routineStatusBanner.textContent = msg;
    routineStatusBanner.className = `px-4 py-2 rounded-lg text-xs font-medium text-center border ${
      isError ? 'bg-rose-950/70 text-rose-300 border-rose-800' : 'bg-emerald-950/70 text-emerald-300 border-emerald-800'
    }`;
    routineStatusBanner.classList.remove('hidden');
    setTimeout(() => routineStatusBanner.classList.add('hidden'), 4000);
  }

  async function populateRoutineAgentSelect(selectedAgentId = null) {
    if (!routineAgentSelect) return;
    const agentsList = buildCreateAgentSelectList(await fetchAllAgents());
    routineAgentSelect.innerHTML = '';
    agentsList.forEach((a) => {
      const opt = document.createElement('option');
      opt.value = a.id;
      opt.textContent = `${a.name || a.id} (${a.id})`;
      if (selectedAgentId && a.id === selectedAgentId) opt.selected = true;
      routineAgentSelect.appendChild(opt);
    });
    if (selectedAgentId && agentsList.some((a) => a.id === selectedAgentId)) {
      routineAgentSelect.value = selectedAgentId;
    }
  }

  function renderChipGroup(root, items, selectedSet) {
    if (!root) return;
    root.innerHTML = '';
    items.forEach(({ label, value }) => {
      const btn = document.createElement('button');
      btn.type = 'button';
      btn.dataset.value = String(value);
      btn.textContent = label;
      const on = selectedSet.has(value);
      btn.className = chipClass(on);
      btn.addEventListener('click', () => {
        const nowOn = !btn.classList.contains('is-selected');
        btn.className = chipClass(nowOn);
        useCronOverride = false;
        schedulePreviewRefresh();
      });
      root.appendChild(btn);
    });
  }

  function ensureScheduleChrome() {
    if (routineMonthPicker && !routineMonthPicker.dataset.ready) {
      renderChipGroup(
        routineMonthPicker,
        MONTH_LABELS.map((label, i) => ({ label, value: i + 1 })),
        new Set(),
      );
      routineMonthPicker.dataset.ready = '1';
    }
    if (routineWeekdayPicker && !routineWeekdayPicker.dataset.ready) {
      renderChipGroup(routineWeekdayPicker, WEEKDAY_CHROME, new Set());
      routineWeekdayPicker.dataset.ready = '1';
    }
    if (routineDomPicker && !routineDomPicker.dataset.ready) {
      renderChipGroup(
        routineDomPicker,
        Array.from({ length: 31 }, (_, i) => ({ label: String(i + 1), value: i + 1 })),
        new Set(),
      );
      routineDomPicker.dataset.ready = '1';
    }
  }

  function applyScheduleRuleToUi(rule) {
    ensureScheduleChrome();
    const r = rule || {};
    const months = new Set(r.months || []);
    const weekdays = new Set(r.weekdays || []);
    const days = new Set(r.days_of_month || []);
    if (routineMonthPicker) {
      routineMonthPicker.querySelectorAll('button.schedule-chip').forEach((b) => {
        b.className = chipClass(months.has(Number(b.dataset.value)));
      });
    }
    if (routineWeekdayPicker) {
      routineWeekdayPicker.querySelectorAll('button.schedule-chip').forEach((b) => {
        b.className = chipClass(weekdays.has(Number(b.dataset.value)));
      });
    }
    if (routineDomPicker) {
      routineDomPicker.querySelectorAll('button.schedule-chip').forEach((b) => {
        b.className = chipClass(days.has(Number(b.dataset.value)));
      });
    }
    if (routineHourInput) routineHourInput.value = r.hour != null ? r.hour : 18;
    if (routineMinuteInput) routineMinuteInput.value = r.minute != null ? r.minute : 0;
