export function getHumanCronPreview(cronStr) {
  const s = (cronStr || '').trim();
  if (!s) return 'Invalid cron expression';
  if (s === '* * * * *') return 'Every minute';
  if (s.startsWith('*/') && s.endsWith('* * * *')) return `Every ${s.slice(2, s.indexOf(' '))} minutes`;
  if (s === '0 * * * *') return 'Every hour at minute 0';
  if (s === '0 */2 * * *') return 'Every 2 hours';
  if (s === '0 8 * * *') return 'Daily at 08:00';
  if (s === '0 12 * * *') return 'Daily at 12:00';
  if (s === '0 9 * * 1-5') return 'Weekdays at 09:00';
  if (s === '0 17 * * 1-5') return 'Weekdays at 17:00';
  if (s === '0 0 * * 0') return 'Weekly on Sunday at 00:00';
  if (s === '0 8 1 * *') return 'Monthly on the 1st at 08:00';
  return `Cron schedule: ${s}`;
}

/** CARD-309: filters over API routine fields (agent_id, enabled, last_run_at, search). */
export function filterRoutinesList(routines, filters = {}) {
  const list = Array.isArray(routines) ? routines : [];
  const q = String(filters.search || '').trim().toLowerCase();
  const agent = String(filters.agent || '').trim();
  const status = String(filters.status || '').trim();
  const lastRan = String(filters.lastRan || '').trim();
  return list.filter((r) => {
    if (agent && String(r.agent_id || '') !== agent) return false;
    if (status === 'active' && !r.enabled) return false;
    if (status === 'paused' && r.enabled) return false;
    const hasRun = Boolean(r.last_run_at);
    if (lastRan === 'never' && hasRun) return false;
    if (lastRan === 'has' && !hasRun) return false;
    if (q) {
      const hay = `${r.name || ''} ${r.id || ''} ${r.prompt || ''} ${r.description || ''} ${r.agent_id || ''} ${r.cron_expression || ''}`.toLowerCase();
      if (!hay.includes(q)) return false;
    }
    return true;
  });
}

/** CARD-310: agents for dropdowns come from /api/agents list only — never routine-derived fallback. */
export function agentsFromApiPayload(payload) {
  if (!Array.isArray(payload)) return [];
  return payload.filter((a) => a && a.id);
}

export function buildFilterAgentOptionsHtml(agents) {
  const ids = [...new Set(agentsFromApiPayload(agents).map((a) => a.id))].sort();
  return '<option value="">All agents</option>' + ids.map((id) => `<option value="${escapeHtml(id)}">${escapeHtml(id)}</option>`).join('');
}

export function buildCreateAgentSelectList(agents) {
  // No hardcoded 3-agent fallback (CARD-310). Empty list stays empty.
  return agentsFromApiPayload(agents);
}

export function readScheduleRuleFromUi(opts = {}) {
  const selected = (root) => {
    if (!root) return [];
    return [...root.querySelectorAll('button.schedule-chip.is-selected')].map((b) => Number(b.dataset.value));
  };
  const months = selected(opts.monthRoot);
  const weekdays = selected(opts.weekdayRoot);
  const days = selected(opts.domRoot);
  const hour = Number(opts.hourInput?.value ?? 0);
  const minute = Number(opts.minuteInput?.value ?? 0);
  const everyN = Number(opts.everyNInput?.value ?? 1);
  const anchor = (opts.anchorInput?.value || '').trim() || null;
  return {
    months: months.length ? months : null,
    weekdays: weekdays.length ? weekdays : null,
    days_of_month: days.length ? days : null,
    hour: Number.isFinite(hour) ? hour : 0,
    minute: Number.isFinite(minute) ? minute : 0,
    every_n_weeks: Number.isFinite(everyN) && everyN >= 1 ? everyN : 1,
    anchor_date: anchor,
  };
}

export function isBuiltinRoutine(routine) {
  if (!routine) return false;
  if (routine.is_builtin !== undefined && routine.is_builtin !== null) {
    return Boolean(routine.is_builtin);
  }
  const BUILTIN_IDS = [
    'routine-sre-health',
    'morning-briefing',
    'daily-sysinfo',
    'nightly-hygiene',
    'hourly-sre-pulse',
    'weekly-note-rollover',
    'skill-eval-sleep',
    'skill-curator',
    'wiki-curation',
    'routine-daily-brief',
    'routine-wiki-prune',
  ];
  return BUILTIN_IDS.includes(routine.id);
}
